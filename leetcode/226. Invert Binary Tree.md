/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public TreeNode invertTree(TreeNode root) {
        return getInvertTree(root);
    }

    private TreeNode getInvertTree(TreeNode cur) {
        if (cur == null) {
            return null;
        }

        TreeNode left = getInvertTree(cur.left);
        TreeNode right = getInvertTree(cur.right);

        cur.left = right;
        cur.right = left;

        return cur;
    }
}
