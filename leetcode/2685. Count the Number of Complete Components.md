class Solution {
    private int[] root, rank, edgeCount; 

    private void union(int x, int y) {
        int xParent = find(x), yParent = find(y); 
        int xRank = root[xParent], yRank = root[yParent]; 

        if (xParent == yParent) {
            edgeCount[xParent]++; 
            return; 
        }

        if (xRank > yRank) {
            root[yParent] = xParent; 
            edgeCount[xParent] += edgeCount[yParent] + 1; 
        } else if (yRank > xRank) {
            root[xParent] = yParent; 
            edgeCount[yParent] += edgeCount[xParent] + 1; 
        } else {
            root[xParent] = yParent; 
            edgeCount[yParent] += edgeCount[xParent] + 1; 
            rank[yParent] += 1; 
        }
    }

    private int find(int x) {
        if (root[x] == x) return x; 
        return root[x] = find(root[x]); // path compression 
    }

    public int countCompleteComponents(int n, int[][] edges) {
        root = new int[n]; 
        rank = new int[n]; 
        edgeCount = new int[n]; 

        for (int i = 0; i < n; i++) {
            root[i] = i; 
            rank[i] = 1; 
            edgeCount[i] = 0; 
        }

        for (int[] e: edges) {
            union(e[0], e[1]); 
        }

        int[] nodeCount = new int[n]; 
        for (int i = 0; i < n; i++) {
            nodeCount[find(i)]++; 
        }

        int cnt = 0; 

        for(int i = 0; i < n; i++) {
            if (nodeCount[i] > 0) {
                // i -> root, it has nodecount 
                if (2 * edgeCount[i] == (nodeCount[i] * (nodeCount[i] - 1))) cnt++; 
            }
        }

        return cnt; 
    }
}
