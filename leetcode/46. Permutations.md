class Solution {
    public List<List<Integer>> permute(int[] nums) {
        List<List<Integer>> result = new ArrayList<>();
        boolean[] isVisit = new boolean[nums.length];

        permutation(nums, new ArrayList<>(), result, isVisit);

        return result;
    }

    private void permutation(int[] nums, List<Integer> list, List<List<Integer>> result, boolean[] isVisit) {
        if (list.size() == nums.length) {
            result.add(new ArrayList<>(list));
            return;
        }

        for (int i = 0; i < nums.length; i++) {
            if (isVisit[i]) continue;

            isVisit[i] = true;
            list.add(nums[i]);
            permutation(nums, list, result, isVisit);
            isVisit[i] = false;
            list.remove(list.size() - 1);
        }
    }
}
