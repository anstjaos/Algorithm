class Solution {
    public int nearestDrone(int[][] drones, int[] target) {
        int min = Integer.MAX_VALUE, result = -1;

        for (int i = 0; i < drones.length; i++) {
            int[] drone = drones[i];
            int dist = Math.abs(target[0] - drone[0]) + Math.abs(target[1] - drone[1]);
            if (dist > drone[2]) continue;

            if (dist < min) {
                min = dist;
                result = i;
            }
        }

        return result;
    }
}
