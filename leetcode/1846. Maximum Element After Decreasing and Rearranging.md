class Solution {
    public int maximumElementAfterDecrementingAndRearranging(int[] arr) {
        int[] count = new int[arr.length + 5];
        for (int num : arr) {
            count[Math.min(arr.length, num)]++;
        }

        int result = 1;
        for (int i = 2; i <= arr.length; i++) {
            result = Math.min(result + count[i], i);
        }

        return result;
    }
}
