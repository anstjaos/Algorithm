class Solution {
    public int lengthOfLongestSubstring(String s) {
        int left = 0, right = 0;
        int result = 0;

        int[] position = new int[129];

        for (char c : s.toCharArray()) {
            left = Math.max(left, position[c]);
            result = Math.max(result, right - left + 1);
            right++;
            position[c] = right;
        }

        return result;
    }
}
