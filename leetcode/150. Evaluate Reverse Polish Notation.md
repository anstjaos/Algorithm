class Solution {
    public int evalRPN(String[] tokens) {
        Stack<Integer> stack = new Stack<>();

        for (String token : tokens) {
            if (isNumeric(token)) {
                stack.push(Integer.parseInt(token));
            } else {
                int second = stack.pop();
                int first = stack.pop();

                int result = switch (token) {
                    case "+" -> first + second;
                    case "-" -> first - second;
                    case "*" -> first * second;
                    case "/" -> first / second;
                    default -> 0;
                };

                stack.push(result);
            }
        }

        return stack.pop();
    }

    private boolean isNumeric(String s) {
        char first = s.charAt(0);

        return ('0' <= first && first <= '9') || (s.length() > 1 && ('0' <= s.charAt(1) && s.charAt(1) <= '9'));
    }
}
