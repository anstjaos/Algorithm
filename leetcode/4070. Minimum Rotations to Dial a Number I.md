class Solution {
    public int minRotations(String s) {
        int cur = 0;
        int result = 0;

        for (char c : s.toCharArray()) {
            int next = (int) (c - '0');
            int diff = Math.abs(next - cur);
            result += Math.min(diff, 10 - diff);
            cur = next;
        }

        return result;
    }
}
