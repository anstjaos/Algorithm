class Solution {
    public int[] dailyTemperatures(int[] temperatures) {
        int length = temperatures.length;

        int[] result = new int[length];
        int[] monoStack = new int[length];
        monoStack[0] = length - 1;
        int top = 0;

        for (int i = length - 2; i >= 0; i--) {
            while (top >= 0 && temperatures[monoStack[top]] <= temperatures[i]) {
                top--;
            }

            monoStack[top + 1] = i;
            if (top >= 0) {
                result[i] = monoStack[top] - i;
            }
            top++;
        }

        return result;
    }
}
