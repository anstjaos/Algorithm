class Solution {
    public String repeatLimitedString(String s, int repeatLimit) {
        int[] count = new int[26];
        for (char c : s.toCharArray()) {
            count[c - 'a']++;
        }

        int backIndex = -1;
        int repeatCount = 0;
        StringBuilder sb = new StringBuilder();
        for (int i = 25; i >= 0; i--) {
            if (count[i] == 0) continue;

            while (repeatCount < repeatLimit && count[i] > 0) {
                sb.append((char) ('a' + i));
                repeatCount++;
                count[i]--;

                if (backIndex != -1 && i != backIndex) {
                    repeatCount = 0;
                    i = backIndex;
                }
            }

            backIndex = -1;
            repeatCount = 0;

            if (count[i] != 0) {
                backIndex = i;
            }
        }

        return sb.toString();
    }
}
