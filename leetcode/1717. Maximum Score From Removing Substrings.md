class Solution {
    private int result = 0;

    public int maximumGain(String s, int x, int y) {
        if (x > y) {
            String next = getScore(s, x, new char[]{'a', 'b'});
            getScore(next, y, new char[]{'b', 'a'});
        } else {
            String next = getScore(s, y, new char[]{'b', 'a'});
            getScore(next, x, new char[]{'a', 'b'});
        }

        return result;
    }

    private String getScore(String s, int score, char[] target) {
        Stack<Character> stack = new Stack<>();
        int result = 0;
        for (char c : s.toCharArray()) {
            if (stack.isEmpty()) {
                stack.push(c);
                continue;
            }

            char top = stack.peek();
            if (top == target[0] && c == target[1]) {
                stack.pop();
                result += score;
            } else {
                stack.push(c);
            }
        }

        StringBuilder sb = new StringBuilder();
        while (!stack.isEmpty()) {
            sb.append(stack.pop());
        }

        this.result += result;

        return sb.reverse().toString();
    }
}
