class Solution {
    private int[] dirX = {-1, 0, 1, 0};
    private int[] dirY = {0, 1, 0, -1};

    public List<List<Integer>> pacificAtlantic(int[][] heights) {
        List<List<Integer>> result = new ArrayList<>();
        int m = heights.length, n = heights[0].length;

        boolean[][] pacific = new boolean[m][n];
        boolean[][] atlantic = new boolean[m][n];

        Queue<int[]> pacificQueue = new LinkedList<>();
        Queue<int[]> atlanticQueue = new LinkedList<>();

        // 태평양 경계: 맨 위 행 + 맨 왼쪽 열
        for (int i = 0; i < m; i++) {
            pacificQueue.add(new int[]{i, 0});
            pacific[i][0] = true;
        }
        for (int j = 0; j < n; j++) {
            pacificQueue.add(new int[]{0, j});
            pacific[0][j] = true;
        }

        // 대서양 경계: 맨 아래 행 + 맨 오른쪽 열
        for (int i = 0; i < m; i++) {
            atlanticQueue.add(new int[]{i, n - 1});
            atlantic[i][n - 1] = true;
        }
        for (int j = 0; j < n; j++) {
            atlanticQueue.add(new int[]{m - 1, j});
            atlantic[m - 1][j] = true;
        }

        bfs(heights, pacificQueue, pacific);
        bfs(heights, atlanticQueue, atlantic);

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (pacific[i][j] && atlantic[i][j]) {
                    result.add(Arrays.asList(i, j));
                }
            }
        }

        return result;
    }

    private void bfs(int[][] heights, Queue<int[]> queue, boolean[][] visited) {
        int m = heights.length, n = heights[0].length;

        while (!queue.isEmpty()) {
            int[] cur = queue.poll();

            for (int i = 0; i < 4; i++) {
                int nextX = cur[0] + dirX[i];
                int nextY = cur[1] + dirY[i];

                if (nextX < 0 || nextY < 0 || nextX >= m || nextY >= n) continue;
                if (visited[nextX][nextY]) continue;
                // 거꾸로 거슬러 올라가는 거라 "낮은 곳 -> 높은 곳"으로 가는 조건
                if (heights[nextX][nextY] < heights[cur[0]][cur[1]]) continue;

                visited[nextX][nextY] = true;
                queue.add(new int[]{nextX, nextY});
            }
        }
    }
}
