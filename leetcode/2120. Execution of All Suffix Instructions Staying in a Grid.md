class Solution {
    public int[] executeInstructions(int n, int[] startPos, String s) {
        int[] result = new int[s.length()];

        for (int i = 0; i < s.length(); i++) {
            String temp = s.substring(i);

            int curX = startPos[0];
            int curY = startPos[1];

            int count = 0;
            for (char c : temp.toCharArray()) {
                if (c == 'L') {
                    curY = curY - 1;
                } else if (c == 'R') {
                    curY = curY + 1;
                } else if (c == 'U') {
                    curX = curX - 1;
                } else {
                    curX = curX + 1;
                }

                if (curX < 0 || curY < 0 || curX >= n || curY >= n) break;
                count++;
            }

            result[i] = count;
        }

        return result;
    }
}
