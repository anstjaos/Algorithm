class Solution {
    public int[] findOrder(int numCourses, int[][] prerequisites) {
        int[] inOrder = new int[numCourses];
        Map<Integer, List<Integer>> outListMap = new HashMap<>();

        for (int[] pre : prerequisites) {
            int to = pre[0];
            int from = pre[1];

            inOrder[to]++;
            List<Integer> outList = outListMap.getOrDefault(from, new ArrayList<>());
            outList.add(to);
            outListMap.put(from, outList);
        }

        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < numCourses; i++) {
            if (inOrder[i] == 0) {
                queue.add(i);
            }
        }

        int[] result = new int[numCourses];
        int index = 0;

        while (!queue.isEmpty()) {
            int cur = queue.poll();
            result[index++] = cur;

            List<Integer> outList = outListMap.getOrDefault(cur, new ArrayList<>());
            for (int next : outList) {
                inOrder[next]--;
                if (inOrder[next] == 0) {
                    queue.add(next);
                }
            }
        }

        for (int i = 0; i < numCourses; i++) {
            if (inOrder[i] != 0) {
                return new int[]{};
            }
        }

        return result;
    }
}
