class Solution {
    public String minWindow(String s, String t) {
        int[] tCount = new int[129];

        for (char c : t.toCharArray()) {
            tCount[c]++;
        }

        int left = 0, right = 0, tSize = t.length();
        int minLength = Integer.MAX_VALUE, startIndex = 0;

        for (; right < s.length(); right++) {
            if (tCount[s.charAt(right)]-- > 0) {
                tSize--;
            }

            while (tSize == 0) {
                if (right - left < minLength) {
                    minLength = right - left;
                    startIndex = left;
                }

                if (tCount[s.charAt(left)]++ == 0) {
                    tSize++;
                }

                left++;
            }
        }

        return minLength != Integer.MAX_VALUE ? s.substring(startIndex, startIndex + minLength + 1) : "";
    }
}
