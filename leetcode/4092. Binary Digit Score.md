class Solution {
    public int calculateScore(int a, int b) {
        int s1 = getBitCount(a);
        int s2 = getBitCount(b);

        if (s1 >= s2) return s1 * s2;

        return s2 / s1;
    }

    private int getBitCount(int num) {
        int count = 0;

        while (num > 0) {
            if ((num & 1) == 1) {
                count++;
            }
            num >>= 1;
        }

        return count;
    }
}
