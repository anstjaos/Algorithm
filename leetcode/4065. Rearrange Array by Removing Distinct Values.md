class Solution {
    public int[] rearrangeArray(int[] nums) {
        int[] count = new int[105];
        for (int num : nums) {
            count[num]++;
        }

        int[] result = new int[nums.length];
        int index = 0;

        while (index != nums.length) {
            for (int i = 1; i < 105; i++) {
                if (count[i] != 0) {
                    result[index++] = i;
                    count[i]--;
                }
            }
        }
        
        return result;
    }
}
