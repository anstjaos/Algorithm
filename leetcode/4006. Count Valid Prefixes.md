class Solution {
    public int countValidPrefixes(String s) {
        int[] count = new int[2];
        int result = 0;
        for (int i = 0; i < s.length(); i++) {
            int num = s.charAt(i) - '0';
            count[num]++;
            if (Math.abs(count[0] - count[1]) <= 1) {
                result++;
            }
        }

        return result;
    }
}
