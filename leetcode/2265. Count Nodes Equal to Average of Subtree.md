/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    private int result = 0;

    public int averageOfSubtree(TreeNode root) {
        getAverageOfSubtree(root);    
        return result;
    }

    private int[] getAverageOfSubtree(TreeNode node) {
        if (node == null) {
            return new int[]{0, 0};
        }

        int[] left = getAverageOfSubtree(node.left);
        int[] right = getAverageOfSubtree(node.right);

        int sum = left[0] + right[0] + node.val;
        int num = left[1] + right[1] + 1;

        if (sum / num == node.val) {
            result++;
        }

        return new int[]{sum, num};
    }
}
