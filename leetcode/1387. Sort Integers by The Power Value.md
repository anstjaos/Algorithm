class Solution {
    class Number {
        int num;
        int count;

        Number(int num, int count) {
            this.num = num;
            this.count = count;
        }
    }

    public int getKth(int lo, int hi, int k) {
        List<Number> list = new ArrayList<>();
        for (int i = lo; i <= hi; i++) {
            int count = getCount(i);
            list.add(new Number(i, count));
        }

        Collections.sort(list, (a, b) -> a.count - b.count);

        return list.get(k - 1).num;
    }

    private int getCount(int num) {
        int count = 0;

        while (num != 1) {
            if ((num & 1) == 1) {
                num = num * 3 + 1;
            } else {
                num /= 2;
            }
            count++;
        }

        return count;
    }
}
