class Solution {
    public int peakValleyDifference(int[] nums) {
        int peakSum = 0, valleySum = 0;

        for (int i = 1; i < nums.length - 1; i++) {
            if (nums[i - 1] > nums[i] && nums[i] < nums[i + 1]) {
                valleySum += nums[i];
            } else if (nums[i - 1] < nums[i] && nums[i] > nums[i + 1]) {
                peakSum += nums[i];
            }
        }

        return peakSum - valleySum;
    }
}
