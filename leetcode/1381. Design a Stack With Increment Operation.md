class CustomStack {
    private int[] stack, inc;
    private int count;

    public CustomStack(int maxSize) {
        stack = new int[maxSize];
        inc = new int[maxSize];
        count = 0;
    }

    public void push(int x) {
        if (count < stack.length) stack[count++] = x;
    }

    public int pop() {
        if (count == 0) return -1;

        int i = --count;
        int res = stack[i] + inc[i];

        if (i > 0) inc[i - 1] += inc[i];
        inc[i] = 0;
        return res;
    }

    public void increment(int k, int val) {
        int i = Math.min(k, count) - 1;
        if (i >= 0) inc[i] += val;
    }
}
