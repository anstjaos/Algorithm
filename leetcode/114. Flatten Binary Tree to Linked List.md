/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public void flatten(TreeNode root) {
        flattenNode(root);
    }

    private void flattenNode(TreeNode cur) {
        if (cur == null) return;

        TreeNode left = cur.left;
        TreeNode right = cur.right;

        flattenNode(left);
        if (left != null) {
            cur.right = left;
            TreeNode temp = left;
            while (temp.right != null) {
                temp = temp.right;
            }
            temp.right = right;
        }
        cur.left = null;
        flattenNode(right);
    }
}
