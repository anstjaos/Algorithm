class Solution {
    public int largestInteger(int n, int s) {
        if (s == 0) return 0;

        int result = 0;
        for (int i = 0; i < n; i++) {
            result *= 10;
            int min = Math.min(s, 9);
            result += min;
            s -= min;
        }

        if (s != 0) return -1;

        return result;
    }
}
