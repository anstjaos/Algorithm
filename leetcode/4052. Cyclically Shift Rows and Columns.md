class Solution {
    public int[][] cyclicShift(int n, int[][] grid, int[] rowShift, int[] colShift) {
        int[][] result = new int[n][n];
        int sr;
        
        for (int r = 0; r < n; r++) {
            for (int c = 0; c < n; c++) {
                result[r][c] = grid[sr = (r + colShift[c]) % n][(c + rowShift[sr]) % n];
            }
        }

        return result;
    }
}
