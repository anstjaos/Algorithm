class Solution {
    private int[] dirX = {-1, 0, 1, 0};
    private int[] dirY = {0, 1, 0, -1};

    public boolean exist(char[][] board, String word) {
        boolean[][] isVisit = new boolean[board.length][board[0].length];

        for (int i = 0; i < board.length; i++) {
            for (int j = 0; j < board[0].length; j++) {
                if (board[i][j] != word.charAt(0)) continue;

                isVisit[i][j] = true;
                if (isExistWord(board, word, 1, i, j, isVisit)) return true;
                isVisit[i][j] = false;
            }
        }
        return false;
    }

    private boolean isExistWord(char[][] board, String word, int index, int x, int y, boolean[][] isVisit) {
        if (index == word.length()) {
            return true;
        }

        for (int i = 0; i < 4; i++) {
            int nextX = x + dirX[i];
            int nextY = y + dirY[i];

            if (nextX < 0 || nextY < 0 || nextX >= board.length || nextY >= board[0].length) continue;
            if (isVisit[nextX][nextY] || board[nextX][nextY] != word.charAt(index)) continue;

            isVisit[nextX][nextY] = true;
            if (isExistWord(board, word, index + 1, nextX, nextY, isVisit)) return true;
            isVisit[nextX][nextY] = false;
        }

        return false;
    }
}
