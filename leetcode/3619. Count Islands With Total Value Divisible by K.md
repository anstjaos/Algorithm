class Solution {
    public int countIslands(int[][] grid, int k) {
        int count = 0;

        for (int i = 0; i < grid.length; i++) {
            for (int j = 0; j < grid[i].length; j++) {
                if (grid[i][j] == 0) continue;

                if (isDivisibleIsland(grid, i, j, k)) {
                    count++;
                }
            }
        }

        return count;
    }

    private boolean isDivisibleIsland(int[][] grid, int x, int y, int k) {
        int[] dirX = {-1, 0, 1, 0}, dirY = {0, 1, 0, -1};
        Queue<int[]> queue = new LinkedList<>();

        queue.add(new int[]{x, y});
        long sum = grid[x][y];
        grid[x][y] = 0;

        while (!queue.isEmpty()) {
            int[] cur = queue.poll();

            for (int i = 0; i < 4; i++) {
                int nextX = cur[0] + dirX[i];
                int nextY = cur[1] + dirY[i];

                if (nextX < 0 || nextY < 0 || nextX >= grid.length || nextY >= grid[0].length) continue;
                if (grid[nextX][nextY] == 0) continue;

                sum += grid[nextX][nextY];
                grid[nextX][nextY] = 0;
                queue.add(new int[]{nextX, nextY});
            }
        }

        return sum % k == 0;
    }
}
