class Solution {
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        int[] inorder = new int[numCourses];
        Map<Integer, List<Integer>> outListMap = new HashMap<>();

        for (int[] prerequiresite : prerequisites) {
            inorder[prerequiresite[0]]++;
            List<Integer> list = outListMap.getOrDefault(prerequiresite[1], new ArrayList<>());
            list.add(prerequiresite[0]);
            outListMap.put(prerequiresite[1], list);
        }

        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < numCourses; i++) {
            if (inorder[i] == 0) queue.add(i);
        }

        while (!queue.isEmpty()) {
            int cur = queue.poll();
            List<Integer> list = outListMap.getOrDefault(cur, new ArrayList<>());
            for (int num : list) {
                inorder[num]--;
                if (inorder[num] == 0) {
                    queue.add(num);
                }
            }
        }

        for (int i = 0; i < numCourses; i++) {
            if (inorder[i] != 0) return false;
        }

        return true;
    }
}
