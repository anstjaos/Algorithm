class Solution {
    public int minDeletions(int[] nums) {
        int result = 0;
        for (int i = 0; i < nums.length; i++) {
            int right = i + 1;
            int count = 0;
            while (right < nums.length && nums[i] == nums[right]) {
                right++;
                count++;
            }

            result += count;
            i = right - 1;
        }

        return result;
    }
}
