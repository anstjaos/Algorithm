class Solution {
    public int countRotations(String s, int k) {
        int length = s.length();
        int result = 0;

        for (int i = 0; i < length; i++) {
            int same = 0;
            for (int j = 0; j < length - 1; j++) {
                if (s.charAt((i + j) % length) == s.charAt((i + j + 1) % length)) {
                    same++;
                }
            }

            if (same == k) result++;
        }

        return result;
    }
}
