class Solution {
    public int countSpecialIntegers(int[] nums) {
        int[] count = new int[105];
        Map<Integer, List<Integer>> numIndexMap = new HashMap<>();

        for (int i = 0; i < nums.length; i++) {
            List<Integer> list = numIndexMap.getOrDefault(nums[i], new ArrayList<>());
            list.add(i);
            count[nums[i]]++;
            numIndexMap.put(nums[i], list);
        }

        int result = 0;

        for (Map.Entry<Integer, List<Integer>> entry : numIndexMap.entrySet()) {
            List<Integer> list = entry.getValue();
            if (list.size() != 3) continue;

            if (list.get(1) - list.get(0) == list.get(2) - list.get(1)) {
                result++;
            }
        }

        return result;
    }
}
