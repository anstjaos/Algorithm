class Solution {
    public int orangesRotting(int[][] grid) {
        int result = rottingOrange(grid);

        for (int i = 0; i < grid.length; i++) {
            for (int j = 0; j < grid[0].length; j++) {
                if (grid[i][j] == 1) {
                    return -1;
                }
            }
        }

        return result;
    }

    private int rottingOrange(int[][] grid) {
        int[] dirX = {-1, 0, 1, 0}, dirY = {0, 1, 0, -1};

        Queue<int[]> queue = new LinkedList<>();
        for (int i = 0; i < grid.length; i++) {
            for (int j = 0; j < grid[0].length; j++) {
                if (grid[i][j] == 2) {
                    queue.add(new int[]{i, j});
                }
            }
        }

        if (queue.isEmpty()) return 0;

        int result = 0;
        while (!queue.isEmpty()) {
            int count = queue.size();

            while (count != 0) {
                int[] cur = queue.poll();
                count--;

                for (int i = 0; i < 4; i++) {
                    int nextX = cur[0] + dirX[i];
                    int nextY = cur[1] + dirY[i];

                    if (nextX < 0 || nextY < 0 || nextX >= grid.length || nextY >= grid[0].length) continue;
                    if (grid[nextX][nextY] != 1) continue;

                    grid[nextX][nextY] = 2;
                    queue.add(new int[]{nextX, nextY});
                }
            }

            result++;
        }

        return result - 1;
    }
}
