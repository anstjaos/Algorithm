class Solution {
    public int partitionString(String s) {
        int result = 0, flag = 0;
        for (char c : s.toCharArray()) {
            int index = (int) (c - 'a');

            if ((flag & (1 << index)) != 0) {
                result++;
                flag = 0;
            }

            flag = flag | (1 << index);
        }

        return result + 1;
    }
}
