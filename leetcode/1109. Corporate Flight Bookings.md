class Solution {
    public int[] corpFlightBookings(int[][] bookings, int n) {
        int[] result = new int[n];
        for (int[] book : bookings) {
            result[book[0] - 1] += book[2];
            if (book[1] < n) {
                result[book[1]] -= book[2];
            }
        }

        for (int i = 1; i < n; i++) {
            result[i] += result[i - 1];
        }

        return result;
    }
}
